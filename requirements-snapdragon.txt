# =============================================================================
# requirements-snapdragon.txt
# Stratum AI — Qualcomm Snapdragon inference backend
# =============================================================================
# REQUIRES: Windows ARM64 machine with Qualcomm Snapdragon SoC
#           (Snapdragon X Elite / X2 Elite or compatible)
#
# Install with:
#   pip install -r requirements-intel.txt        # common dependencies first
#   pip install -r requirements-snapdragon.txt   # Snapdragon-specific additions
#
# =============================================================================

# ── ONNX Runtime with QNN Execution Provider ──────────────────────────────────
# Official package — ARM64 Windows only, provides QNNExecutionProvider
# https://pypi.org/project/onnxruntime-qnn/
onnxruntime-qnn

# ── ONNX Runtime GenAI (token-level streaming on Snapdragon) ─────────────────
# Provides true token streaming for generative models.
# Optional — if not installed, backend uses a compatibility streaming layer.
# https://pypi.org/project/onnxruntime-genai/
onnxruntime-genai

# ── Base ONNX Runtime (may be installed as dependency of the above) ───────────
onnxruntime

# ── Model download helper ─────────────────────────────────────────────────────
huggingface_hub

# Model download (Snapdragon path)

#   (Download pre-optimised ONNX assets for Snapdragon X Elite)
#

#   huggingface-cli download qualcomm/Qwen3-4B \
#       --local-dir models/Qwen3-4B-onnx
#
# Then set environment variable:
#   SNAPDRAGON_MODEL_PATH=models/Qwen3-4B-onnx
